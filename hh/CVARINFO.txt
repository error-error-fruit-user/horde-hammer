// Horde Hammer's settings. Server settings are the host's, user settings each player's own.
//
// Every server setting is -1 (or "" for text) until someone sets it, which means "the game's
// default": the value in the game's HH_Defaults subclass, or the library's own in HH_Defaults
// (zscript/hh/settings.zs). The engine allows no other way for a game to change a library's
// defaults (redefining a CVar is a fatal error, and DEFCVARS only loads from the IWAD). Switches
// are numbers: 0 off, 1 on. Code reads them through HH_Settings, never by name.

// Players
server int hh_start_points = -1;                 // points a player starts with
server int hh_respawn_points = -1;               // the least a player has after respawning at a new round
server int hh_max_weapons = -1;                  // weapons a player can carry, not counting melee and drinks
server int hh_max_equipment = -1;              // pieces of equipment a player can carry
server int hh_perk_limit = -1;                   // perks a player can hold at once, 0 for no limit
server float hh_player_speed = -1;               // player speed, 0 to leave the player class's own
server int hh_health = -1;                       // health without Juggernog
server int hh_jugg_health = -1;                  // health with Juggernog
server int hh_health_regen = -1;                 // health comes back after a while without damage
server int hh_regen_delay = -1;                  // tics without damage before healing starts
server int hh_regen_rate = -1;                   // health healed per tic
server int hh_weapon_switching = -1;             // Horde Hammer handles the weapon slot keys (turn off for mods with their own)
server int hh_remove_items = -1;                 // remove items that appear in the world (monster drops, map items)
server int hh_kick_enabled = -1;                        // the kick key works
server int hh_kick_damage = -1;                 
server int hh_hud_before_start = -1;            // the HUD shows before round 1 (a game with an intro turns it off)
server int hh_drop_weapon_cost = -1;             // points to drop a weapon (double tap the drop key)
server int hh_pad_layout = -1;                   // offer each player the zombies controller layout, once (controller.zs)

// Rounds
server int hh_auto_start = -1;                   // start round 1 by itself (a game with an intro turns it off and starts it itself)
server int hh_start_delay = -1;                  // tics before round 1 when it starts by itself
server int hh_round_delay = -1;                  // tics between the last zombie dying and repulsion
server int hh_repulsion_time = -1;               // tics of repulsion before the next round; the repulse key starts it again
server int hh_zombie_health_base = -1;           // zombie health in round 1
server int hh_zombie_health_per_round = -1;      // added each round
server int hh_zombie_count_base = -1;            // zombies in round 1
server int hh_zombie_count_per_round = -1;       // added each round
server int hh_zombie_count_per_player = -1;      // added each round for every player after the first
server float hh_zombie_speed_per_round = -1;     // added to zombie speed each round
server float hh_zombie_speed_max = -1;           // the most speed rounds can give, 0 for no limit
server float hh_zombie_slow_speed = -1;          // a slow zombie's speed against its own
server int hh_zombie_slow_chance = -1;           // percent of zombies that are slow
server float hh_zombie_fast_speed = -1;          // a fast zombie's speed against its own
server int hh_zombie_fast_per_round = -1;        // percent more of them fast each round after the first
server int hh_zombie_fast_max = -1;              // the most percent of them fast
server int hh_max_horde = -1;                    // zombies alive at once
server int hh_spawn_delay_min = -1;              // tics between spawns
server int hh_spawn_delay_max = -1;             
server string hh_zombie_class = "";              // what spawners spawn (an HH_ZombieSpawner subclass picks one at random)

// Points
server int hh_points_hit = -1;                  
server int hh_points_kill = -1;                 
server int hh_points_headshot = -1;             
server int hh_points_melee = -1;                
server int hh_points_bottle = -1;               

// The mystery box
server int hh_box_cost = -1;                    
server int hh_box_fire_sale_cost = -1;          
server int hh_box_min_spins = -1;                // spins before the box can move
server int hh_box_move_chance = -1;              // after that, it moves 1 time in this many
server int hh_box_pickup_time = -1;              // tics to take the weapon
server int hh_box_respin_cost = -1;
server int hh_box_kick_drop_cost = -1;
server int hh_box_push = -1;
server int hh_roster_reveal_time = -1;
server int hh_roster_page_time = -1;
server int hh_roster_idle_time = -1;
server int hh_roster_hold_time = -1;
server int hh_unlock_cost = -1;

// Wall buys
server int hh_wall_ammo_divisor = -1;            // ammo costs the weapon's cost divided by this
server int hh_wall_upgraded_ammo_cost = -1;     

// Drops
server int hh_drops_per_round = -1;             
server int hh_drop_chance = -1;                  // a kill drops a power-up 1 time in this many
server int hh_drop_lifetime = -1;                // tics a drop stays on the floor
server int hh_powerup_duration = -1;             // tics a power-up lasts
server int hh_kaboom_points = -1;               

// Going down
server int hh_bleed_out_time = -1;               // tics a downed player waits for a revive
server int hh_revive_time = -1;                  // tics to revive someone
server int hh_quick_revive_time = -1;            // with Quick Revive
server int hh_risen_time = -1;
server int hh_risen_rounds = -1;
server int hh_solo_last_stand_time = -1;         // tics before solo Quick Revive brings the player back
server int hh_solo_revives = -1;                 // solo Quick Revive can be bought this many times
server int hh_solo_quick_revive_cost = -1;      

// Each player's own
user bool hh_toggled_zoom = true;               // zoom stays in until pressed again
user bool hh_pad_layout_set = false;            // this player was offered the controller layout (false offers it again)
user noarchive string hh_unlocked = "";    // never saved: unlocks last the session

// The HUD, each player's own: where each part is (0 to 1 across and down the screen), how big it
// is, and whether it shows. The menu has a reset.
user float hh_hud_scale = 1.0;
user float hh_hud_alpha = 1.0;
user bool hh_hud_points_show = true;
user float hh_hud_points_x = 0.03;
user float hh_hud_points_y = 0.04;
user float hh_hud_points_scale = 1.0;
user bool hh_hud_round_show = true;
user float hh_hud_round_x = 0.97;
user float hh_hud_round_y = 0.04;
user float hh_hud_round_scale = 1.0;

user bool hh_hud_messages_show = true;
